   23: #include <cstdint>
   24: #include <cstdio>
   25: #include <cstdlib>
   26: 
   27: static uint32_t live_handles = 0, handle_token = 0;
   28: static bool backend_dead = false;
   29: static void verify_cleanup() {
   30:     if (live_handles != 0) std::_Exit(92);
   31:     std::fputs("DG_PROBE_CLEAN\n", stderr);
   32: }
   33: // Registered before the extension's static destructors, so this runs last.
   34: __attribute__((constructor)) static void initialize_probe() {
   35:     std::atexit(verify_cleanup);
   36: }
   37: static void stop_backend() { backend_dead = true; }
   38: extern "C" uint32_t cublasLtCreate(void** handle) {
   39:     std::atexit(stop_backend);
   40:     *handle = &handle_token;
   41:     ++live_handles;
   42:     std::fputs("DG_PROBE_CREATE\n", stderr);
   43:     return 0;
   44: }
   45: extern "C" uint32_t cublasLtDestroy(void* handle) {
   46:     if (backend_dead) std::_Exit(91);
   47:     if (handle != &handle_token or live_handles != 1) std::_Exit(93);
   48:     --live_handles;
   49:     std::fputs("DG_PROBE_DESTROY\n", stderr);
   50:     return 0;
   51: }
   52: '''

三个 magic exit code 的语义（测试体靠它们判定）：
  _Exit(91): cublasLtDestroy 在 backend_dead 之后被调用 → 退出期 use-after-free
  _Exit(92): verify_cleanup 时 live_handles != 0 → handle 泄漏（未被销毁）
  _Exit(93): cublasLtDestroy 收到非本探针 handle 或 live_handles != 1 → 重复/陌生销毁

状态机：
  initialize_probe()  __attribute__((constructor))  →  std::atexit(verify_cleanup)
  cublasLtCreate  →  std::atexit(stop_backend); ++live_handles; 打印 DG_PROBE_CREATE
  cublasLtDestroy →  backend_dead?  _Exit(91);  handle 不符? _Exit(93); --live_handles; 打印 DG_PROBE_DESTROY
  退出时 verify_cleanup 最后跑（constructor 里先注册，故在扩展静态析构之后）→ 打印 DG_PROBE_CLEAN 或 _Exit(92)
